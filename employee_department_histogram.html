import io

csv_data = (
    "employee_id,department,region,performance_score,years_experience,satisfaction_rating\n"
    "EMP001,Finance,Asia Pacific,61.4,1,4.1\n"
    "EMP002,R&D,Africa,70.09,7,3.6\n"
    "EMP003,Sales,Middle East,72.33,14,3.6\n"

df = pd.read_csv(io.StringIO(csv_data))

# Plot department distribution

plt.ylabel("Count")
